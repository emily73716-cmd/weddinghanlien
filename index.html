        /* 單個音符合成 */

        function playSynthesizedNote(freq, startTime, duration, type = 'triangle', volume = 0.18) {

            if (!audioCtx || !freq) return;

            try {

                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();


                osc.type = type;

                osc.frequency.setValueAtTime(freq, startTime);


                // 柔和溫馨的琴音包絡

                gain.gain.setValueAtTime(0, startTime);

                gain.gain.linearRampToValueAtTime(volume, startTime + 0.04);

                gain.gain.exponentialRampToValueAtTime(0.0001, startTime + duration * 1.2);


                osc.connect(gain);

                gain.connect(audioCtx.destination);


                osc.start(startTime);

                osc.stop(startTime + duration * 1.2);

            } catch (e) {

                console.warn("Audio synthesis notice:", e);

            }

        }

        function playWeddingMarchSequence(index = 0) {

            if (!isMusicPlaying || !audioCtx) return;


            if (index >= MENDELSSOHN_SCORE.length) {

                // 樂曲結束後停頓 1.5 秒自動循環

                musicTimeoutId = setTimeout(() => {

                    playWeddingMarchSequence(0);

                }, 1500);

                return;

            }


            const item = MENDELSSOHN_SCORE[index];

            const now = audioCtx.currentTime;

            const tempoFactor = 0.85; 

            const durSeconds = item.dur * tempoFactor;


            // 播放主旋律

            if (item.note && NOTE_FREQS[item.note]) {

                playSynthesizedNote(NOTE_FREQS[item.note], now, durSeconds, 'triangle', 0.22);

                // 八度泛音

                playSynthesizedNote(NOTE_FREQS[item.note] * 2, now, durSeconds, 'sine', 0.04);

            }


            // 播放底層和聲

            if (item.chord && item.chord.length > 0) {

                item.chord.forEach(chordNote => {

                    const freq = NOTE_FREQS[chordNote];

                    if (freq) {

                        playSynthesizedNote(freq, now, durSeconds * 1.1, 'sine', 0.07);

                    }

                });

            }


            // 預約下一個音符

            musicTimeoutId = setTimeout(() => {

                playWeddingMarchSequence(index + 1);

            }, durSeconds * 1000);

        }
